Git: land + merge (only after sign-off)

git add deploy/LAPTOP-DEPLOY.md
git commit -m "Slice 15: add beginner laptop deploy-day guide"

# on the laptop - checkout the HITL branch, NOT master
git clone https://github.com/kaijudon/SUBAY.git /opt/subay
cd /opt/subay && git checkout agent/slice-15

# after the sign-off table is signed - the real "done"
git checkout master && git merge --no-ff agent/slice-15
